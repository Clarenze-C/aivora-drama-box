import os, sys
sys.path.insert(0, "src")
from microdrama.pipeline.canary import refuse_non_loopback_dsn
def t(dsn, **env):
    saved = {k: os.environ.pop(k, None) for k in ("PGHOST","PGHOSTADDR","PGSERVICE")}
    os.environ.update(env)
    try:
        refuse_non_loopback_dsn(dsn); r = "PASS"
    except Exception as e:
        r = "REFUSED(" + type(e).__name__ + ")"
    for k in env: os.environ.pop(k, None)
    for k, v in saved.items():
        if v is not None: os.environ[k] = v
    print(f"{r:28} {dsn!r} {env or ''}")
